// PORTAL SIMONSPORTS — SINAL DIÁRIO CONSOLIDADO PARA O YOUTUBE
//
// PADRÃO V19 — VIGENTE A PARTIR DE 01/08/2026
// • 1 vídeo completo reunindo todas as loterias sorteadas no dia.
// • No máximo 1 Short diário.
// • Máximo de 2 publicações por dia.
// • Com apenas uma loteria, o Short só é gerado quando houver destaque especial.
//
// IMPORTANTE
// Este Apps Script NÃO manda mais publicar uma modalidade isoladamente.
// Ele apenas avisa o GitHub Actions de que existe resultado novo. O workflow
// "Publicação diária consolidada no YouTube" consulta a planilha, confere a aba
// Calendário_Loterias e só publica quando todas as modalidades previstas para a
// data estiverem disponíveis.
//
// CONFIGURAÇÃO ÚNICA
// Em Apps Script > Configurações do projeto > Propriedades do script, cadastre:
// GITHUB_TOKEN_DISPATCH = token do GitHub com acesso ao repositório
// portalsimonsports/twitter-loteria-bot e permissão para repository_dispatch.
//
// USO DIRETO NO CÓDIGO QUE GRAVA O RESULTADO
// Depois de concluir todas as escritas da linha, execute:
// SpreadsheetApp.flush();
// DISPARAR_VERIFICACAO_YOUTUBE_DIARIA(dataSorteio, loteria, concurso);
//
// A função antiga DISPARAR_PUBLICACAO_YOUTUBE_IMEDIATA foi mantida apenas para
// compatibilidade. Ela agora localiza a data do concurso e chama o fluxo diário.

const YOUTUBE_GITHUB_OWNER_ = 'portalsimonsports';
const YOUTUBE_GITHUB_REPO_ = 'twitter-loteria-bot';
const YOUTUBE_GITHUB_EVENTO_ = 'resultado_loteria_disponivel';
const YOUTUBE_PLANILHA_ID_ = '16NcdSwX6q_EQ2XjS1KNIBe6C3Piq-lCBgA38TMszXCI';
const YOUTUBE_ABA_RESULTADOS_ = 'ImportadosBlogger2';
const YOUTUBE_COLUNA_PUBLICADO_DIARIO_ = 'Publicado_Youtube_Diario';
const YOUTUBE_DATA_INICIO_DIARIO_ = '01/08/2026';
const YOUTUBE_REDISPARO_MINUTOS_ = 15;
const YOUTUBE_FUSO_HORARIO_ = 'America/Sao_Paulo';

/**
 * Envia um sinal de verificação do lote diário ao GitHub Actions.
 * O sinal não autoriza publicação individual e não ignora modalidades faltantes.
 */
function DISPARAR_VERIFICACAO_YOUTUBE_DIARIA(dataSorteio, loteria, concurso) {
  const dataReferencia = NORMALIZAR_DATA_YOUTUBE_(dataSorteio);
  const nomeLoteria = String(loteria || '').trim();
  const numeroConcurso = String(concurso || '').trim();

  if (!dataReferencia) {
    throw new Error(
      'Data do sorteio inválida para a verificação diária do YouTube: ' +
      String(dataSorteio || '')
    );
  }

  SpreadsheetApp.flush();

  const token = PropertiesService.getScriptProperties()
    .getProperty('GITHUB_TOKEN_DISPATCH');

  if (!token) {
    throw new Error(
      'Propriedade GITHUB_TOKEN_DISPATCH não configurada nas Propriedades do script.'
    );
  }

  const endpoint =
    'https://api.github.com/repos/' +
    encodeURIComponent(YOUTUBE_GITHUB_OWNER_) + '/' +
    encodeURIComponent(YOUTUBE_GITHUB_REPO_) +
    '/dispatches';

  const payload = {
    event_type: YOUTUBE_GITHUB_EVENTO_,
    client_payload: {
      modo: 'diario_consolidado',
      data: dataReferencia,
      loteria_origem: nomeLoteria,
      concurso_origem: numeroConcurso,
      maximo_publicacoes_dia: 2,
      origem: 'Google Apps Script',
      planilha_id: YOUTUBE_PLANILHA_ID_,
      aba: YOUTUBE_ABA_RESULTADOS_,
      enviado_em: new Date().toISOString()
    }
  };

  const resposta = UrlFetchApp.fetch(endpoint, {
    method: 'post',
    contentType: 'application/json',
    payload: JSON.stringify(payload),
    headers: {
      Authorization: 'Bearer ' + token,
      Accept: 'application/vnd.github+json',
      'X-GitHub-Api-Version': '2022-11-28',
      'User-Agent': 'Portal-SimonSports-Apps-Script'
    },
    muteHttpExceptions: true
  });

  const codigo = resposta.getResponseCode();
  if (codigo !== 204) {
    throw new Error(
      'GitHub recusou o sinal diário. HTTP ' + codigo + ': ' +
      resposta.getContentText().slice(0, 1000)
    );
  }

  console.log(
    '[YOUTUBE DIÁRIO] Verificação solicitada para ' + dataReferencia +
    (nomeLoteria ? ' após ' + nomeLoteria : '') +
    (numeroConcurso ? ' — concurso ' + numeroConcurso : '')
  );

  return true;
}

/**
 * Compatibilidade com códigos que ainda chamam a função antiga.
 * Não publica mais o concurso isoladamente.
 */
function DISPARAR_PUBLICACAO_YOUTUBE_IMEDIATA(loteria, concurso, dataSorteio) {
  const nomeLoteria = String(loteria || '').trim();
  const numeroConcurso = String(concurso || '').trim();

  if (!nomeLoteria) {
    throw new Error('Loteria não informada para o sinal diário do YouTube.');
  }
  if (!numeroConcurso) {
    throw new Error('Concurso não informado para o sinal diário do YouTube.');
  }

  const dataReferencia =
    NORMALIZAR_DATA_YOUTUBE_(dataSorteio) ||
    LOCALIZAR_DATA_RESULTADO_YOUTUBE_(nomeLoteria, numeroConcurso);

  if (!dataReferencia) {
    throw new Error(
      'Não foi possível localizar a data de ' + nomeLoteria +
      ' — concurso ' + numeroConcurso + '.'
    );
  }

  return DISPARAR_VERIFICACAO_YOUTUBE_DIARIA(
    dataReferencia,
    nomeLoteria,
    numeroConcurso
  );
}

/**
 * Procura datas com resultados completos e ainda sem marca final diária.
 * Cada data gera no máximo um repository_dispatch dentro do intervalo definido.
 */
function VERIFICAR_RESULTADOS_PENDENTES_YOUTUBE() {
  const planilha = SpreadsheetApp.openById(YOUTUBE_PLANILHA_ID_);
  const aba = planilha.getSheetByName(YOUTUBE_ABA_RESULTADOS_);
  if (!aba) {
    throw new Error('Aba não encontrada: ' + YOUTUBE_ABA_RESULTADOS_);
  }

  const ultimaLinha = aba.getLastRow();
  const ultimaColuna = aba.getLastColumn();
  if (ultimaLinha < 2 || ultimaColuna < 1) {
    return 0;
  }

  const valores = aba.getRange(1, 1, ultimaLinha, ultimaColuna).getDisplayValues();
  const cabecalhos = valores[0].map(function(valor) {
    return NORMALIZAR_CABECALHO_YOUTUBE_(valor);
  });

  const colunaLoteria = LOCALIZAR_COLUNA_YOUTUBE_(cabecalhos, ['loteria', 'produto']);
  const colunaConcurso = LOCALIZAR_COLUNA_YOUTUBE_(cabecalhos, ['concurso']);
  const colunaData = LOCALIZAR_COLUNA_YOUTUBE_(cabecalhos, ['data', 'data sorteio']);
  const colunaNumeros = LOCALIZAR_COLUNA_YOUTUBE_(cabecalhos, [
    'numeros', 'números', 'descricao', 'descrição'
  ]);
  const colunaUrl = LOCALIZAR_COLUNA_YOUTUBE_(cabecalhos, ['url', 'link']);
  const colunaPublicadoDiario = LOCALIZAR_COLUNA_YOUTUBE_(cabecalhos, [
    YOUTUBE_COLUNA_PUBLICADO_DIARIO_,
    'publicado youtube diario',
    'publicado_youtube_diario'
  ]);

  if (
    colunaLoteria < 0 || colunaConcurso < 0 || colunaData < 0 ||
    colunaNumeros < 0 || colunaUrl < 0
  ) {
    throw new Error(
      'A aba não contém todas as colunas obrigatórias para a verificação diária do YouTube.'
    );
  }

  const inicioNumero = DATA_NUMERO_YOUTUBE_(YOUTUBE_DATA_INICIO_DIARIO_);
  const hoje = Utilities.formatDate(
    new Date(),
    YOUTUBE_FUSO_HORARIO_,
    'dd/MM/yyyy'
  );
  const hojeNumero = DATA_NUMERO_YOUTUBE_(hoje);
  const limiteInferior = Math.max(1, valores.length - 500);
  const datasPendentes = {};

  for (let indice = valores.length - 1; indice >= limiteInferior; indice--) {
    const linha = valores[indice];
    const loteria = String(linha[colunaLoteria] || '').trim();
    const concurso = String(linha[colunaConcurso] || '').trim();
    const data = NORMALIZAR_DATA_YOUTUBE_(linha[colunaData]);
    const numeros = String(linha[colunaNumeros] || '').trim();
    const url = String(linha[colunaUrl] || '').trim();
    const marcadorDiario = colunaPublicadoDiario >= 0
      ? String(linha[colunaPublicadoDiario] || '').trim()
      : '';

    if (!loteria || !concurso || !data || !numeros || !url) {
      continue;
    }

    const dataNumero = DATA_NUMERO_YOUTUBE_(data);
    if (!dataNumero || dataNumero < inicioNumero || dataNumero > hojeNumero) {
      continue;
    }

    if (MARCADOR_FINAL_YOUTUBE_DIARIO_(marcadorDiario)) {
      if (!datasPendentes[data]) {
        datasPendentes[data] = {
          finalizado: true,
          loteria: loteria,
          concurso: concurso
        };
      } else {
        datasPendentes[data].finalizado = true;
      }
      continue;
    }

    if (!datasPendentes[data]) {
      datasPendentes[data] = {
        finalizado: false,
        loteria: loteria,
        concurso: concurso
      };
    }
  }

  const propriedades = PropertiesService.getScriptProperties();
  const agora = Date.now();
  const intervalo = YOUTUBE_REDISPARO_MINUTOS_ * 60 * 1000;
  let disparos = 0;

  Object.keys(datasPendentes)
    .sort(function(a, b) {
      return DATA_NUMERO_YOUTUBE_(a) - DATA_NUMERO_YOUTUBE_(b);
    })
    .forEach(function(data) {
      const grupo = datasPendentes[data];
      if (!grupo || grupo.finalizado) {
        return;
      }

      const assinatura = CRIAR_ASSINATURA_YOUTUBE_('DIARIO|' + data);
      const chave = 'YOUTUBE_DISPATCH_DIARIO_' + assinatura;
      const ultimoDisparo = Number(propriedades.getProperty(chave) || 0);

      if (ultimoDisparo && agora - ultimoDisparo < intervalo) {
        return;
      }

      DISPARAR_VERIFICACAO_YOUTUBE_DIARIA(
        data,
        grupo.loteria,
        grupo.concurso
      );
      propriedades.setProperty(chave, String(agora));
      disparos++;
    });

  console.log('[YOUTUBE DIÁRIO] Datas pendentes sinalizadas: ' + disparos);
  return disparos;
}

/**
 * Instala uma verificação de reserva a cada 15 minutos.
 * O GitHub Actions já possui verificação própria a cada 5 minutos.
 */
function INSTALAR_GATILHO_YOUTUBE_DIARIO() {
  REMOVER_GATILHOS_YOUTUBE_DIARIO_();
  ScriptApp.newTrigger('VERIFICAR_RESULTADOS_PENDENTES_YOUTUBE')
    .timeBased()
    .everyMinutes(15)
    .create();

  console.log('[YOUTUBE DIÁRIO] Gatilho de reserva instalado a cada 15 minutos.');
}

/** Compatibilidade com o nome antigo do instalador. */
function INSTALAR_GATILHO_YOUTUBE_IMEDIATO() {
  return INSTALAR_GATILHO_YOUTUBE_DIARIO();
}

/** Remove os gatilhos antigos e atuais deste módulo. */
function REMOVER_GATILHOS_YOUTUBE_DIARIO_() {
  const funcoes = [
    'VERIFICAR_RESULTADOS_PENDENTES_YOUTUBE'
  ];

  ScriptApp.getProjectTriggers().forEach(function(gatilho) {
    if (funcoes.indexOf(gatilho.getHandlerFunction()) >= 0) {
      ScriptApp.deleteTrigger(gatilho);
    }
  });
}

/** Compatibilidade interna com o nome antigo. */
function REMOVER_GATILHOS_YOUTUBE_IMEDIATO_() {
  return REMOVER_GATILHOS_YOUTUBE_DIARIO_();
}

function LOCALIZAR_DATA_RESULTADO_YOUTUBE_(loteria, concurso) {
  const planilha = SpreadsheetApp.openById(YOUTUBE_PLANILHA_ID_);
  const aba = planilha.getSheetByName(YOUTUBE_ABA_RESULTADOS_);
  if (!aba || aba.getLastRow() < 2) {
    return '';
  }

  const valores = aba.getDataRange().getDisplayValues();
  const cabecalhos = valores[0].map(function(valor) {
    return NORMALIZAR_CABECALHO_YOUTUBE_(valor);
  });
  const colunaLoteria = LOCALIZAR_COLUNA_YOUTUBE_(cabecalhos, ['loteria', 'produto']);
  const colunaConcurso = LOCALIZAR_COLUNA_YOUTUBE_(cabecalhos, ['concurso']);
  const colunaData = LOCALIZAR_COLUNA_YOUTUBE_(cabecalhos, ['data', 'data sorteio']);

  if (colunaLoteria < 0 || colunaConcurso < 0 || colunaData < 0) {
    return '';
  }

  const loteriaAlvo = NORMALIZAR_CABECALHO_YOUTUBE_(loteria);
  const concursoAlvo = String(concurso || '').replace(/\D+/g, '');
  const limiteInferior = Math.max(1, valores.length - 500);

  for (let indice = valores.length - 1; indice >= limiteInferior; indice--) {
    const linha = valores[indice];
    const loteriaLinha = NORMALIZAR_CABECALHO_YOUTUBE_(linha[colunaLoteria]);
    const concursoLinha = String(linha[colunaConcurso] || '').replace(/\D+/g, '');

    if (loteriaLinha === loteriaAlvo && concursoLinha === concursoAlvo) {
      return NORMALIZAR_DATA_YOUTUBE_(linha[colunaData]);
    }
  }

  return '';
}

function MARCADOR_FINAL_YOUTUBE_DIARIO_(valor) {
  const texto = NORMALIZAR_CABECALHO_YOUTUBE_(valor);
  return texto.indexOf('publicado youtube diario v19') >= 0;
}

function LOCALIZAR_COLUNA_YOUTUBE_(cabecalhos, alternativas) {
  const alvos = alternativas.map(function(valor) {
    return NORMALIZAR_CABECALHO_YOUTUBE_(valor);
  });

  for (let indice = 0; indice < cabecalhos.length; indice++) {
    if (alvos.indexOf(cabecalhos[indice]) >= 0) {
      return indice;
    }
  }
  return -1;
}

function NORMALIZAR_CABECALHO_YOUTUBE_(valor) {
  return String(valor || '')
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .replace(/[_-]+/g, ' ')
    .replace(/\s+/g, ' ')
    .trim()
    .toLowerCase();
}

function NORMALIZAR_DATA_YOUTUBE_(valor) {
  if (valor instanceof Date && !isNaN(valor.getTime())) {
    return Utilities.formatDate(
      valor,
      YOUTUBE_FUSO_HORARIO_,
      'dd/MM/yyyy'
    );
  }

  const texto = String(valor || '').trim();
  let partes = texto.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
  if (partes) {
    return String(Number(partes[1])).padStart(2, '0') + '/' +
      String(Number(partes[2])).padStart(2, '0') + '/' + partes[3];
  }

  partes = texto.match(/^(\d{4})-(\d{1,2})-(\d{1,2})$/);
  if (partes) {
    return String(Number(partes[3])).padStart(2, '0') + '/' +
      String(Number(partes[2])).padStart(2, '0') + '/' + partes[1];
  }

  return '';
}

function DATA_NUMERO_YOUTUBE_(valor) {
  const data = NORMALIZAR_DATA_YOUTUBE_(valor);
  const partes = data.match(/^(\d{2})\/(\d{2})\/(\d{4})$/);
  if (!partes) {
    return 0;
  }
  return Number(partes[3] + partes[2] + partes[1]);
}

function CRIAR_ASSINATURA_YOUTUBE_(valor) {
  const bytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    String(valor || ''),
    Utilities.Charset.UTF_8
  );

  return Utilities.base64EncodeWebSafe(bytes)
    .replace(/=+$/g, '')
    .slice(0, 32);
}
